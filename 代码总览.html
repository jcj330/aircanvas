/* ============================================================
 * 模块 3：CDN 多源容错加载器
 * 动态创建 <script> 标签按需加载（页面打开时不预先加载模型，
 * 只有点击"启动"后才开始，避免启动卡顿）。
 * jsdelivr 失败自动切换 unpkg。
 * ============================================================ */
function loadScriptOnce(url, timeoutMs = 20000) {
  return new Promise((resolve, reject) => {
    const s = document.createElement('script');
    const timer = setTimeout(() => { s.remove(); reject(new Error('加载超时')); }, timeoutMs);
    s.src = url;
    s.onload = () => { clearTimeout(timer); resolve(); };
    s.onerror = () => { clearTimeout(timer); s.remove(); reject(new Error('网络加载失败')); };
    document.head.appendChild(s);
  });
}

/** 依次尝试各 CDN 源加载 hands.js，全部失败抛出异常 */
async function loadHandsLibrary() {
  for (const base of CONFIG.CDN_BASES) {
    try {
      await loadScriptOnce(base + '/hands.js');
      if (typeof window.Hands === 'function') return base; // 加载成功且全局对象可用
    } catch (e) { /* 换下一个源 */ }
  }
  throw new Error('所有 CDN 均不可用');
}

async function startCamera() {
  // 安全上下文检测：部分浏览器在 file:// 下直接不暴露摄像头 API
  if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
    const e = new Error('当前浏览器在此打开方式下无法访问摄像头');
    e.tip = '请改用 Chrome 或 Edge 直接双击打开本文件；\n若仍不行，可按 README 说明启动本地服务器后访问 localhost。';
    throw e;
  }
  let stream;
  try {
    stream = await navigator.mediaDevices.getUserMedia({
      video: { width: { ideal: 640 }, height: { ideal: 480 }, facingMode: 'user' },
      audio: false
    });
  } catch (err) {
    // 按错误类型给出可操作的解决建议
    const map = {
      NotAllowedError: '摄像头权限被拒绝。请点击地址栏左侧的相机/锁形图标，将摄像头设为「允许」后刷新页面重试。',
      NotFoundError: '未检测到摄像头设备。请确认电脑已连接摄像头并启用。',
      NotReadableError: '摄像头被其他程序占用（如腾讯会议 / Zoom 等）。请关闭占用程序后重试。'
    };
    err.tip = map[err.name] || ('摄像头打开失败：' + (err.message || err.name));
    throw err;
  }
  video.srcObject = stream;
  await video.play();                 // muted+autoplay 环境下 play() 不会被策略拦截
  await new Promise(r => {             // 等待第一帧就绪，避免黑帧送入模型
    if (video.readyState >= 2) return r();
    video.onloadeddata = () => r();
  });
}

/* ============================================================
 * 模块 5：手势识别器
 * MediaPipe Hands 输出 21 个手部关键点（归一化坐标）。
 * 关键点索引：0手腕 4拇指尖 8食指尖 12中指尖 16无名指尖 20小指尖
 *             5/9/13/17 分别为四指根部关节
 *
 * 判定思路（对远近、手掌大小不敏感）：
 *   - 掌宽参考 = dist(手腕0, 中指根9)，用于拇指判定归一化；
 *   - 四指伸直：指尖到手腕的距离 > (PIP关节到手腕距离 × 1.12)
 *     （弯曲时指尖向掌心收拢，距手腕变近）；
 *   - 拇指伸直：拇指尖到中指根距离 > 掌宽 × 0.7
 *     （握拳时拇指横贴在四指上，距离显著变小）。
 * 稳定器：连续 STABLE_FRAMES 帧判定一致才切换确认手势，
 *         消除单帧抖动造成的误触发。
 * ============================================================ */
function dist2d(a, b) { const dx = a.x - b.x, dy = a.y - b.y; return Math.hypot(dx, dy); }

/** 由 21 个关键点判定每根手指的伸直状态 */
function fingerStates(lm) {
  const wrist = lm[0];
  const palm = dist2d(lm[0], lm[9]) || 1e-6; // 掌宽参考值（防除零）
  const ext = (tip, pip) => dist2d(lm[tip], wrist) > dist2d(lm[pip], wrist) * 1.12;
  return {
    index:  ext(8, 6),
    middle: ext(12, 10),
    ring:   ext(16, 14),
    pinky:  ext(20, 18),
    thumb:  dist2d(lm[4], lm[9]) > palm * 0.7
  };
}

/** 将手指状态归类为四种手势之一 */
function classifyGesture(f) {
  if (f.index && !f.middle && !f.ring && !f.pinky) return 'INDEX';   // 只伸食指 → 绘制
  if (!f.index && !f.middle && !f.ring && !f.pinky && !f.thumb) return 'FIST'; // 握拳 → 暂停
  if (f.index && f.middle && f.ring && f.pinky && f.thumb) return 'PALM';      // 张开手掌 → 停止
  return 'OTHER';                                                    // 其他 → 抬笔悬停
}

/** 手势稳定器：连续 N 帧一致才确认，防止边界抖动 */
const stabilizer = {
  lastRaw: null, count: 0, stable: null,
  feed(raw) {
    if (raw === this.lastRaw) this.count++;
    else { this.lastRaw = raw; this.count = 1; }
    if (this.count >= CONFIG.STABLE_FRAMES && raw !== this.stable) this.stable = raw;
    return this.stable;
  },
  reset() { this.lastRaw = null; this.count = 0; this.stable = null; }
};

/* ============================================================
 * 模块 6：状态机
 * 状态：IDLE 待机(无手) | DRAW 绘制 | PAUSE 暂停抬笔 | ERASE 橡皮擦 | STOP 停止
 * 转移表（g = 稳定后的手势）：
 *   IDLE  --食指--> DRAW        DRAW  --握拳--> PAUSE
 *   PAUSE --握拳保持1s--> 切换工具(画笔/橡皮擦)
 *   PAUSE --食指--> DRAW/ERASE    任意 --张开手掌--> STOP
 *   STOP  --食指--> DRAW         任意 --手离开1.5s--> IDLE
 *   任意 --其他手势--> PAUSE(仅抬笔，不改状态徽章)
 * ============================================================ */
function setState(next) {
  if (state.phase === next) return;
  if (state.phase === 'DRAW' || state.phase === 'ERASE') draw.endStroke(); // 离开绘制状态必须抬笔
  state.phase = next;
  updateBadge();
}

/** 每帧驱动状态机；返回当前是否处于"笔在纸上"的绘制状态 */
function stepStateMachine(gesture, hasHand, now) {
  // --- 无手 1.5 秒 → 待机 ---
  if (!hasHand) {
    if (state.phase !== 'IDLE' && now - state.lastHandTime > CONFIG.NOHAND_MS) setState('IDLE');
    return false;
  }
  state.lastHandTime = now;

  switch (gesture) {
    case 'INDEX': {
      // 伸食指：按当前工具进入对应绘制状态
      const target = state.tool === 'eraser' ? 'ERASE' : 'DRAW';
      setState(target);
      return state.phase === target; // 只有刚进入（或已处于）绘制状态才落笔
    }
    case 'FIST': {
      setState('PAUSE');
      // 握拳计时：持续 1 秒 → 切换画笔/橡皮擦（可来回切换）
      if (!state.fistHoldStart) { state.fistHoldStart = now; state.fistHoldFired = false; }
      else if (!state.fistHoldFired && now - state.fistHoldStart >= CONFIG.FIST_HOLD_MS) {
        state.fistHoldFired = true;
        setTool(state.tool === 'pen' ? 'eraser' : 'pen', true);
      }
      return false;
    }
    case 'PALM': setState('STOP'); return false;   // 张开手掌 → 停止待机
    default:                                     // OTHER：其他手势一律抬笔，进入暂停态
      setState('PAUSE');
      draw.endStroke();
      return false;
  }
}

/* ============================================================
 * 模块 7：One Euro 自适应滤波器（笔迹平滑防抖核心）
 * 原理：一阶低通滤波，但截止频率随速度自适应——
 *   指尖移动慢 → 截止频率低 → 滤波强 → 消除手抖；
 *   指尖移动快 → 截止频率高 → 滤波弱 → 保留快速挥笔的流畅轨迹。
 * 论文：Casiez et al., "1€ Filter", CHI 2012
 * x、y 两个维度分别独立滤波。
 * ============================================================ */
class LowPass {
  constructor() { this.y = null; }
  reset() { this.y = null; }
  filter(x, alpha) { this.y = (this.y === null) ? x : alpha * x + (1 - alpha) * this.y; return this.y; }
}

class OneEuroFilter {
  constructor(minCutoff, beta, dCutoff) {
    this.minCutoff = minCutoff; this.beta = beta; this.dCutoff = dCutoff;
    this.xF = new LowPass(); this.dxF = new LowPass();
    this.tPrev = null; this.xPrev = 0;
  }
  reset() { this.xF.reset(); this.dxF.reset(); this.tPrev = null; }
  /** 滤波器的平滑系数：截止频率越低（tau 越大）α 越小 → 平滑越强 */
  alpha(cutoff, dt) { const tau = 1 / (2 * Math.PI * cutoff); return 1 / (1 + tau / dt); }
  filter(x, tMs) {
    if (this.tPrev === null) { this.tPrev = tMs; this.xPrev = x; this.xF.reset(); return x; }
    const dt = Math.max((tMs - this.tPrev) / 1000, 1e-3); // 秒，防除零
    this.tPrev = tMs;
    // 估计速度并对其做低通（抑制噪声引起的伪速度）
    const dx = (x - this.xPrev) / dt; this.xPrev = x;
    const edx = this.dxF.filter(dx, this.alpha(this.dCutoff, dt));
    // 自适应截止频率：速度越快截止越高（beta 加权）
    const cutoff = this.minCutoff + this.beta * Math.abs(edx);
    return this.xF.filter(x, this.alpha(cutoff, dt));
  }
}

/** 二维坐标封装：对 x/y 分别滤波；抬笔后 reset 防止新旧笔画间"飞线" */
const smoother = {
  fx: new OneEuroFilter(CONFIG.EURO.minCutoff, CONFIG.EURO.beta, CONFIG.EURO.dCutoff),
  fy: new OneEuroFilter(CONFIG.EURO.minCutoff, CONFIG.EURO.beta, CONFIG.EURO.dCutoff),
  reset() { this.fx.reset(); this.fy.reset(); },
  next(x, y, t) { return { x: this.fx.filter(x, t), y: this.fy.filter(y, t) }; }
};

/* ============================================================
 * 模块 8：绘图引擎
 * 画布为透明层叠加在镜像视频上方，逻辑分辨率 640×480。
 * 笔迹绘制：二次贝塞尔中点插值——以上一段中点为起点、
 *   关键点为控制点、下一段中点为终点，天然圆滑无折角。
 * 橡皮擦：globalCompositeOperation = 'destination-out'，
 *   粗线扫过即从已画内容中"减去"。
 * 落笔反馈：起点画圆点 + 涟漪动画。
 * ============================================================ */
const draw = {
  pts: [],        // 当前笔画最近的关键点（贝塞尔需要保留最近 3 个）
  active: false,

  /** 橡皮擦粗细 = 画笔粗细 × 4（夹在 20~90 之间，保证擦得动） */
  eraserWidth() { return Math.min(90, Math.max(20, state.penSize * 4)); },

  beginStroke(p, erasing) {
    history.pushSnapshot();                    // 新笔画开始前快照 → 供撤销
    this.active = true; this.erasing = erasing;
    this.pts = [p];
    this.dot(p, erasing);                      // 落笔圆点
    spawnRipple(p, erasing);                   // 落笔涟漪动画（UI 反馈）
  },

  extendStroke(p) {
    if (!this.active) return;
    this.pts.push(p);
    const n = this.pts.length;
    if (n >= 3) {
      const [a, b, c] = [this.pts[n - 3], this.pts[n - 2], this.pts[n - 1]];
      // 中点插值：起点=ab中点，控制点=b，终点=bc中点
      this.seg({ x: (a.x + b.x) / 2, y: (a.y + b.y) / 2 }, b, { x: (b.x + c.x) / 2, y: (b.y + c.y) / 2 });
      this.pts = this.pts.slice(-2);           // 只保留最近 2 点，控制内存
    } else if (n === 2) {
      this.seg(this.pts[0], this.pts[0], { x: (this.pts[0].x + p.x) / 2, y: (this.pts[0].y + p.y) / 2 });
    }
  },

  endStroke() {
    if (!this.active) { this.pts = []; return; }
    // 收尾：把最后半段补画到末端点
    const n = this.pts.length;
    if (n >= 2) this.seg(this.pts[n - 2], this.pts[n - 1], this.pts[n - 1]);
    this.active = false; this.pts = [];
  },

  /** 画一段贝塞尔曲线（或擦除段） */
  seg(p0, p1, p2) {
    const erasing = this.erasing;
    pctx.save();
    if (erasing) {
      pctx.globalCompositeOperation = 'destination-out';
      pctx.lineWidth = this.eraserWidth();
    } else {
      pctx.strokeStyle = state.color;
      pctx.lineWidth = state.penSize;
      pctx.shadowColor = state.color; pctx.shadowBlur = 8;  // 霓虹发光效果
    }
    pctx.lineCap = 'round'; pctx.lineJoin = 'round';
    pctx.beginPath();
    pctx.moveTo(p0.x, p0.y);
    pctx.quadraticCurveTo(p1.x, p1.y, p2.x, p2.y);
    pctx.stroke();
    pctx.restore();
  },

  /** 落笔圆点 */
  dot(p, erasing) {
    pctx.save();
    if (erasing) {
      pctx.globalCompositeOperation = 'destination-out';
      pctx.beginPath(); pctx.arc(p.x, p.y, this.eraserWidth() / 2, 0, Math.PI * 2); pctx.fill();
    } else {
      pctx.fillStyle = state.color;
      pctx.shadowColor = state.color; pctx.shadowBlur = 10;
      pctx.beginPath(); pctx.arc(p.x, p.y, state.penSize / 2, 0, Math.PI * 2); pctx.fill();
    }
    pctx.restore();
  }
};

/* ============================================================
 * 模块 9：撤销 / 重做（ImageData 快照双栈）
 * 快照时机：每个笔画"开始前"压栈（含清空前）——
 *   撤销一次即回到上一笔画之前的状态，语义直观。
 * 新笔画开始时清空重做栈（标准编辑器行为）。
 * ============================================================ */
const history = {
  undoStack: [], redoStack: [],
  snapshot() { return pctx.getImageData(0, 0, CONFIG.CANVAS_W, CONFIG.CANVAS_H); },
  pushSnapshot() {
    this.undoStack.push(this.snapshot());
    if (this.undoStack.length > CONFIG.HISTORY_MAX) this.undoStack.shift(); // 超限丢弃最旧
    this.redoStack.length = 0;
    updateHistoryButtons();
  },
  undo() {
    if (!this.undoStack.length) return toast('没有可撤销的操作', 'warn');
    this.redoStack.push(this.snapshot());
    pctx.putImageData(this.undoStack.pop(), 0, 0);
    updateHistoryButtons();
  },
  redo() {
    if (!this.redoStack.length) return toast('没有可重做的操作', 'warn');
    this.undoStack.push(this.snapshot());
    pctx.putImageData(this.redoStack.pop(), 0, 0);
    updateHistoryButtons();
  }
};

/** 按状态更新指尖光标样式 */

function initToolbar() {
  // 色板
  PALETTE.forEach((c, i) => {
    const s = document.createElement('button');
    s.className = 'swatch' + (i === 0 ? ' active' : '');
    s.style.background = c;
    s.title = c;
    s.onclick = () => {
      state.color = c;
      document.querySelectorAll('.swatch').forEach(x => x.classList.remove('active'));
      s.classList.add('active');
      if (state.tool === 'eraser') setTool('pen'); // 选颜色自动回到画笔
      updateCursorStyle();
    };
    $('palette').appendChild(s);
  });

  // 粗细滑块
  $('sizeSlider').oninput = e => {
    state.penSize = +e.target.value;
    $('sizeLabel').textContent = state.penSize + ' px';
  };

  // 工具按钮
  $('btnPen').onclick = () => setTool('pen');
  $('btnEraser').onclick = () => setTool('eraser');

  // 撤销 / 重做（含键盘快捷键）
  $('btnUndo').onclick = () => history.undo();
  $('btnRedo').onclick = () => history.redo();
  document.addEventListener('keydown', e => {
    if (!(e.ctrlKey || e.metaKey)) return;
    if (e.key.toLowerCase() === 'z' && !e.shiftKey) { e.preventDefault(); history.undo(); }
    if (e.key.toLowerCase() === 'y' || (e.shiftKey && e.key.toLowerCase() === 'z')) { e.preventDefault(); history.redo(); }
  });

  // 清空：两段式确认（第一次点击变红提示，3 秒内再点一次才执行）
  let clearArmed = 0, clearTimer = null;
  $('btnClear').onclick = () => {
    const btn = $('btnClear');
    if (clearArmed && Date.now() - clearArmed < 3000) {
      history.pushSnapshot();                                   // 清空前同样可撤销
      pctx.clearRect(0, 0, CONFIG.CANVAS_W, CONFIG.CANVAS_H);
      clearArmed = 0; btn.classList.remove('confirm');
      clearTimeout(clearTimer);
      toast('画布已清空（可撤销）');
    } else {
      clearArmed = Date.now();
      btn.classList.add('confirm');
      toast('再点一次「清空」确认', 'warn');
      clearTimeout(clearTimer);
      clearTimer = setTimeout(() => { btn.classList.remove('confirm'); clearArmed = 0; }, 3000);
    }
  };

  // 暂停 / 恢复追踪
  $('btnPause').onclick = () => {
    state.running = !state.running;
    const icon = $('btnPause');
    if (state.running) { setState('IDLE'); startLoop(); toast('已恢复追踪'); icon.classList.remove('active'); }
    else { draw.endStroke(); setState('PAUSE'); toast('已暂停追踪（点击恢复）', 'warn'); icon.classList.add('active'); }
  };

  // 保存 PNG
  $('btnSave').onclick = savePNG;
  updateHistoryButtons();
}

/* ============================================================
 * 模块 12：主流程编排
 * 启动流程：点击启动 → 开摄像头 → 加载 MediaPipe（CDN 容错）
 *           → 首帧识别成功后进入主界面
 * 主循环：rAF + hands.send 逐帧识别
 *   onResults：关键点 → 手势稳定 → 状态机 → 滤波 → 绘制/抬笔
 * ============================================================ */
let hands = null;          // MediaPipe Hands 实例
let sendBusy = false;      // 防止上一帧未处理完就送下一帧
let firstResult = true;    // 首帧标志（用于关闭加载动画）
let lastT = performance.now(), fpsEMA = 0;

/** MediaPipe 每帧识别结果回调（真正的"主循环体"） */
function onResults(results) {
  const now = performance.now();

  // --- FPS 统计（指数滑动平均） ---
  const dt = now - lastT; lastT = now;
  const f = 1000 / Math.max(dt, 1);
  fpsEMA = fpsEMA ? fpsEMA * 0.9 + f * 0.1 : f;
  fpsEl.textContent = Math.round(fpsEMA) + ' FPS';

  // --- 首帧到达：关闭加载页，进入主界面 ---
  if (firstResult) {
    firstResult = false;
    $('overlay').classList.add('hide');
    toast('将手伸到镜头前，伸出食指即可开始绘画');
  }

  const lms = results.multiHandLandmarks;
  const hasHand = lms && lms.length > 0;

  if (!hasHand) {
    stepStateMachine(null, false, now);
    state.fistHoldStart = 0; state.fistHoldFired = false; // 手丢失时重置握拳计时，防止回来误触发
    cursorEl.style.display = 'none';
    // 丢帧抖动容错：连续约 0.6 秒无手才显示提示，避免闪烁
    if (state.lastHandTime && now - state.lastHandTime > 600) noHandTip.classList.add('show');
    return;
  }
  noHandTip.classList.remove('show');

  const lm = lms[0];                          // 只追踪一只手（性能考虑）
  const gesture = stabilizer.feed(classifyGesture(fingerStates(lm)));
  const drawing = stepStateMachine(gesture, true, now);

  // --- 食指尖端(8号点)坐标：视频做了镜像显示，x 坐标需要取反 ---
  const raw = { x: (1 - lm[8].x) * CONFIG.CANVAS_W, y: lm[8].y * CONFIG.CANVAS_H };

  if (drawing) {
    const p = smoother.next(raw.x, raw.y, now);
    if (!draw.active) {
      smoother.reset();                       // 新笔画从当前点起步，防止从旧位置"飞线"
      const p0 = smoother.next(raw.x, raw.y, now);
      draw.beginStroke(p0, state.tool === 'eraser');
    } else {
      draw.extendStroke(p);
    }
    moveCursor(p);
  } else {
    smoother.reset();                         // 抬笔期间持续重置，恢复落笔时不跳变
    moveCursor(raw);
  }
  updateCursorStyle();

  // --- 手势离开握拳状态时，重置握拳计时器 ---
  if (gesture !== 'FIST') { state.fistHoldStart = 0; state.fistHoldFired = false; }
}

# 空中画板 AirCanvas

基于电脑摄像头手部追踪的"空中绘画"互动网页。伸出一根食指，对准镜头在空中比划，指尖轨迹就会实时绘制成画——握拳暂停、握拳保持 1 秒切换橡皮擦、张开手掌停止待机。

## 运行环境

| 项目 | 要求 |
|---|---|
| 浏览器 | Chrome 或 Edge（推荐，版本 80+）；Firefox 可用但体验略差 |
| 硬件 | 一个可用的摄像头（笔记本自带即可） |
| 网络 | 首次运行需联网加载手部追踪模型（约 3 MB，之后有浏览器缓存） |
| 安装 | **无需安装任何软件、无需构建步骤** |

## 怎么运行

### 方式一：直接双击（推荐）

双击打开 `index.html`（默认会用 Chrome / Edge 打开）：

1. 点击页面上的「启动体验」按钮；
2. 浏览器弹出摄像头权限询问时，选择「允许」；
3. 等待模型加载完成（首次约几秒到十几秒，取决于网速）；
4. 把手伸到镜头前，伸出食指即可开始绘画。

### 方式二：本地服务器（备用）

若你的浏览器在直接双击打开时无法唤起摄像头（个别浏览器会拦截 file:// 页面的摄像头权限），用下面任意一种方式启动本地服务：

**有 Python**（在项目文件夹打开终端）：

```bash
python -m http.server 8080
```

然后浏览器访问 `http://localhost:8080`

**有 VS Code**：安装 Live Server 插件 → 右键 `index.html` → Open with Live Server

## 手势操作对照

| 手势 | 效果 |
|---|---|
| ☝️ 伸食指 | 绘制（当前工具为画笔）或擦除（当前工具为橡皮擦） |
| ✊ 握拳 | 暂停抬笔，可移动手而不画线 |
| ✊ 握拳保持 1 秒 | 在画笔 / 橡皮擦之间来回切换 |
| 🖐️ 张开手掌 | 停止待机 |
| 手离开画面 | 1.5 秒后进入待机 |

## 工具栏功能

画笔/橡皮切换 · 10 色色板 · 粗细调节（2–30 px）· 撤销/重做（Ctrl+Z / Ctrl+Y，最多 25 步）· 两段式确认清空 · 暂停/恢复追踪 · 导出 PNG

## 常见问题

| 现象 | 原因与解决 |
|---|---|
| 弹出权限询问后点错成"拒绝" | 点击地址栏左侧的相机/锁形图标 → 摄像头设为「允许」→ 刷新页面 |
| 提示"当前浏览器在此打开方式下无法访问摄像头" | 该浏览器拦截了本地文件的摄像头权限：改用 Chrome/Edge 双击打开，或用方式二启动本地服务器 |
| 提示"摄像头被其他程序占用" | 关闭腾讯会议、Zoom 等正在使用摄像头的程序后重试 |
| 模型加载失败 | 检查网络连接后刷新页面重试（模型来自 CDN） |
| 识别不到手 / 手势乱跳 | 保证光线充足、手完整出现在画面中、距离镜头 40–80 cm；避免逆光 |
| 笔迹抖动明显 | 已内置 One Euro 自适应滤波，若仍明显可在代码中调大 `CONFIG.EURO.minCutoff`（如 1.5） |

## 实现结构速览（index.html 内 12 个模块）

1. UI 结构与样式 —— 深色玻璃拟态界面
2. 全局配置 `CONFIG` —— 所有可调参数集中一处
3. CDN 多源容错加载器 —— jsdelivr 失败自动切 unpkg，按需加载不卡启动

5. 手势识别器 —— 21 关键点掌宽归一化判定 + 连续 4 帧稳定器
6. 状态机 —— 绘制/暂停/橡皮擦/停止/待机的转移逻辑
7. One Euro 滤波器 —— 低速强滤波防抖、高速弱滤波保流畅
8. 绘图引擎 —— 二次贝塞尔中点插值平滑笔迹、落笔涟漪反馈
9. 撤销/重做 —— ImageData 快照双栈（上限 25 张）
10. 工具栏交互 —— 色板、滑块、两段式清空确认、快捷键
